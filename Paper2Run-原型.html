from model import ResNet18
from dataset import load_cifar10

CFG = {
    "epochs": 120,
    "batch_size": 128,      # RUN#2 修复: 512 → 128
    "lr": 0.05,            # RUN#1 修复: 1.0 → 0.05
    "momentum": 0.9,
    "weight_decay": 5e-4,
    "seed": 42,
}

def train():
    torch.manual_seed(CFG["seed"])
    model = ResNet18().cuda()
    train_loader, val_loader = load_cifar10(CFG["batch_size"])
    opt = torch.optim.SGD(model.parameters(), lr=CFG["lr"],
                          momentum=CFG["momentum"],
                          weight_decay=CFG["weight_decay"])
    sched = CosineAnnealingLR(opt, CFG["epochs"])
    for epoch in range(CFG["epochs"]):
        model.train()
        for x, y in train_loader:
            loss = F.cross_entropy(model(x.cuda()), y.cuda())
            opt.zero_grad(); loss.backward(); opt.step()
        sched.step()
        if epoch % 20 == 0:
            acc = evaluate(model, val_loader)
            print(f"epoch={epoch} loss={loss.item():.4f} acc={acc:.2%}")
    torch.save(model.state_dict(), "resnet18_cifar10.pt")

if __name__ == "__main__":
    train()
